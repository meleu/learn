Include <stdint.h>. Avoid int (only guaranteed ≥ 16 bits) and plain char (signedness is implementation-defined) for hardware state.

Wraparound & promotion

  • Unsigned arithmetic wraps modulo 2n. It is well-defined (cppreference).
  • Signed overflow is undefined behavior. Keep emulator state unsigned.
  • Integer promotion: operands narrower than int are widened to int before arithmetic (cppreference). Truncation happens only on assignment back to a narrow type.
uint8_t v = 0xFF; v += 1;          // 0
uint8_t z = 0;    z -= 1;          // 255

// add with carry: compute wide, then store narrow, flag LAST
int sum   = c8.V[x] + c8.V[y];
c8.V[x]   = sum;                   // truncated to 8 bits
c8.V[0xF] = sum > 0xFF;           // 1 if carry

Machine state

Langhoff, Specifications

typedef struct {
    uint8_t  memory[4096];     // 0x000–0xFFF; programs load at 0x200
    uint8_t  V[16];            // V0..VF (VF = flag)
    uint16_t I;                // index register
    uint16_t pc;               // program counter
    uint16_t stack[16];        // return addresses
    uint8_t  sp;               // stack pointer
    uint8_t  delay_timer;      // −1 at 60 Hz until 0
    uint8_t  sound_timer;      // beeps while > 0
    uint8_t  display[64 * 32]; // 1 byte per pixel
    uint8_t  keypad[16];       // keys 0x0–0xF
} Chip8;

static Chip8 c8;               // static ⇒ zero-initialized
